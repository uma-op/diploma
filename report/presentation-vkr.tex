\begin{frame}{Изоморфизм Карри-Говарда}

Изоморфизм Карри-Говарда устанавливает соответствие между логикой и программированием

\begin{align*}
  I = \lambda x.x &: A \rightarrow A \\
  K = \lambda xy.x &: A \rightarrow (B \rightarrow A) \\
  \lambda x.x(\lambda y.y) &: ((A \rightarrow A) \rightarrow B) \rightarrow B
\end{align*}
\end{frame}

\begin{frame}{Супертермы}
  \scriptsize
  \begin{align*}
    \langle T_1, \ldots, T_n \rangle & = \lambda p. p T_1 \ldots T_n \\
    Proj(n, i, p) & = p \lambda x_1 \ldots x_n . x_i\\
    Insert(n, i, p, e) & = \langle \mathrm{Proj}(n, 1, p),\mathrm{Proj}(n, 2, p),\ldots \underbrace{e}_{i\text{-я позиция}}\ldots,\mathrm{Proj}(n, n, p) \rangle \\
    Sum(n, i, e) &= \lambda c_1 \ldots c_n . c_i e \\
    Case(e, [c_1]b_1, \ldots, [c_n]b_n) &= e(\lambda c_1.b_1)\ldots(\lambda c_n.b_n) \\
    K(e) &= \lambda x.e
  \end{align*}

\end{frame}

  \item SYJ201. $p_1, \ldots, p_m$ - атомы, $c(N) = p_1 \land \ldots \land p_m$

  \item SYJ207. $\bigwedge_{i=1}^{m} ((p_i \leftrightarrow p_{i+1}) \rightarrow c(N)) \rightarrow (p_0 \vee \neg p_0 \vee c(N))$
  \item fresh-tail. $(\ldots((A \rightarrow p_1) \rightarrow p_1) \ldots \rightarrow p_m) \rightarrow p_m$
\end{itemize}
